### Ejemplos

El siguiente ejemplo demuestra cómo implementar un proveedor de formas de palabras personalizado.

```csharp
public class SimpleWordFormsProvider : IWordFormsProvider
{
    public string[] GetWordForms(string word)
    {
        List<string> result = new List<string>();

        // Supongamos que la palabra de entrada está en plural, entonces agregamos el singular.
        if (word.Length > 2 &&
            word.EndsWith("es", StringComparison.InvariantCultureIgnoreCase))
        {
            result.Add(word.Substring(0, word.Length - 2));
        }
        if (word.Length > 1 &&
            word.EndsWith("s", StringComparison.InvariantCultureIgnoreCase))
        {
            result.Add(word.Substring(0, word.Length - 1));
        }

        // Luego supongamos que la palabra de entrada está en singular, agregamos el plural.
        if (word.Length > 1 &&
            word.EndsWith("y", StringComparison.InvariantCultureIgnoreCase))
        {
            result.Add(word.Substring(0, word.Length - 1) + "is");
        }
        result.Add(word + "s");
        result.Add(word + "es");
        // Todas las reglas están implementadas en la clase EnglishWordFormsProvider.

        return result.ToArray();
    }
}
```

El siguiente ejemplo demuestra cómo establecer un proveedor de formas de palabras personalizado para su uso.

```csharp
string indexFolder = @"c:\MyIndex\";
string documentsFolder = @"c:\MyDocuments\";
  
// Creando un índice en la carpeta especificada
Index index = new Index(indexFolder);
  
// Indexando documentos de la carpeta especificada
index.Add(documentsFolder);
 
// Estableciendo la instancia del proveedor de formas de palabras personalizado
index.Dictionaries.WordFormsProvider = new SimpleWordFormsProvider();
 
// Creando una instancia de opciones de búsqueda
SearchOptions options = new SearchOptions();
options.UseWordFormsSearch = true; // Enabling search for word forms
  
// Buscando en el índice
SearchResult result = index.Search("relative", options);
  
// Se pueden encontrar las siguientes palabras:
// relativo
// parientes
```
